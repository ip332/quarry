cmake_minimum_required(VERSION 3.20)
project(quarry_truck_multi_configuration LANGUAGES C CXX)
find_package(Quarry CONFIG REQUIRED)
set(rev_a "${CMAKE_CURRENT_SOURCE_DIR}/../schemas/configuration.brd")
set(rev_b "${CMAKE_CURRENT_SOURCE_DIR}/rev_b/configuration.brd")
set(cpp_dir "${CMAKE_CURRENT_BINARY_DIR}/generated/cpp")
quarry_generate_cpp(SCHEMA "${rev_a}" OUTPUT_DIR "${cpp_dir}" NAME_PREFIX rev_a
    SCHEMA_ID_VARIABLE id_a OUT_FILES cpp_a)
quarry_generate_cpp(SCHEMA "${rev_b}" OUTPUT_DIR "${cpp_dir}" NAME_PREFIX rev_b
    SCHEMA_ID_VARIABLE id_b OUT_FILES cpp_b)
if(NOT id_a STREQUAL "9200ca494f010801b830d0857cf66dde" OR id_a STREQUAL id_b)
    message(FATAL_ERROR "unexpected complete truck configuration identities")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/schema_ids.txt" "${id_a}\n${id_b}\n")
add_executable(truck_multi_cpp owning.cpp ${cpp_a} ${cpp_b})
target_include_directories(truck_multi_cpp PRIVATE "${cpp_dir}")
target_link_libraries(truck_multi_cpp PRIVATE Quarry::runtime)
target_compile_features(truck_multi_cpp PRIVATE cxx_std_20)

# There is no installed C/C-storage generation helper. Follow the explicit
# custom-command pattern used by the other public C examples.
file(GLOB schemas CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../schemas/*.brd"
    "${CMAKE_CURRENT_SOURCE_DIR}/rev_b/*.brd")
function(fixed_bindings variable directory root prefix storage)
    if(storage)
        set(flags --language cpp --cpp-storage c)
        set(header hpp)
        set(source cpp)
    else()
        set(flags --language c)
        set(header h)
        set(source c)
    endif()
    set(outputs)
    foreach(ns truck truck/engine truck/common truck/cms truck/brakes truck/transmission)
        list(APPEND outputs "${directory}/${prefix}/${ns}.generated.${header}"
            "${directory}/${prefix}/${ns}.generated.${source}")
    endforeach()
    add_custom_command(OUTPUT ${outputs}
        COMMAND "$<TARGET_FILE:Quarry::schema_compiler>" ${flags} --name-prefix "${prefix}"
            --output-directory "${directory}" "${root}"
        DEPENDS ${schemas} Quarry::schema_compiler VERBATIM)
    set(${variable} "${outputs}" PARENT_SCOPE)
endfunction()
foreach(kind c c_storage)
    set(directory "${CMAKE_CURRENT_BINARY_DIR}/generated/${kind}")
    if(kind STREQUAL c_storage)
        set(storage TRUE)
        set(main c_storage.cpp)
    else()
        set(storage FALSE)
        set(main plain.c)
    endif()
    fixed_bindings(files_a "${directory}" "${rev_a}" rev_a ${storage})
    fixed_bindings(files_b "${directory}" "${rev_b}" rev_b ${storage})
    add_executable(truck_multi_${kind} ${main} ${files_a} ${files_b})
    target_include_directories(truck_multi_${kind} PRIVATE "${directory}")
    target_link_libraries(truck_multi_${kind} PRIVATE Quarry::runtime_c)
    set_target_properties(truck_multi_${kind} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON
        C_EXTENSIONS OFF CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
endforeach()
foreach(target truck_multi_cpp truck_multi_c truck_multi_c_storage)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
